## 📝 技术实现细节

### 图片上传

```typescript
// 使用 ImageUpload 组件
<ImageUpload
  value={imageUrls}
  onChange={setImageUrls}
  maxFiles={5}
  disabled={submitting}
/>

// 保存时包含图片数据
images: imageUrls.map((url, index) => ({
  url,
  alt: formData.name,
  sortOrder: index,
}))
```

### 批量操作

```typescript
// 使用 BulkActions 组件
<BulkActions
  selectedIds={selectedProducts}
  actions={[
    { label: '批量发布', value: 'publish', icon: CheckCircle },
    { label: '批量下架', value: 'unpublish', icon: XCircle },
    { label: '批量删除', value: 'delete', icon: Trash2, variant: 'destructive' },
  ]}
  onAction={handleBulkAction}
/>

// 批量操作处理
const handleBulkAction = async (action: string) => {
  for (const productId of selectedProducts) {
    await fetch(`/api/products/${productId}`, {
      method: 'PATCH',
      body: JSON.stringify({ status: newStatus }),
    });
  }
};
```

### 状态快速切换

```typescript
// 快速切换按钮
<Button
  size="sm"
  variant={product.status === 'PUBLISHED' ? 'outline' : 'default'}
  onClick={() => toggleProductStatus(product.id)}
>
  {product.status === 'PUBLISHED' ? '下架' : '发布'}
</Button>

// 切换处理
const toggleProductStatus = async (productId: string) => {
  const newStatus = product.status === 'PUBLISHED' ? 'DRAFT' : 'PUBLISHED';
  await fetch(`/api/products/${productId}`, {
    method: 'PATCH',
    body: JSON.stringify({ status: newStatus }),
  });
};
```
